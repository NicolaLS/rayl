package xyz.lilsus.rayl.providers.nwc.integration.nwc

import java.net.URI
import java.net.http.HttpClient
import java.net.http.HttpRequest
import java.net.http.HttpResponse
import kotlin.test.AfterTest
import kotlin.test.Test
import kotlin.test.assertIs
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.runBlocking
import kotlinx.serialization.json.Json
import kotlinx.serialization.json.jsonObject
import kotlinx.serialization.json.jsonPrimitive
import xyz.lilsus.raylsuite.core.model.WalletStartMode

/**
 * Rayl's NWC wallet against Alby Hub, the most used NWC wallet service, through the NWC faucet
 * (faucet.nwc.dev: Alby Hub sub-wallets with test balances, qa/test-plan.md §6.9). It checks
 * that Rayl reads a real wallet's responses as the stubs assume (qa/stub-assumptions.md, A6).
 * Failure cases that need control over the wallet use the local NWC stack instead.
 * Run: `./gradlew :providers:nwc:integration:nwc:walletIntegrationTest`.
 */
class NwcFaucetWalletIntegrationTest {
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Default)
    private val wallets = mutableListOf<NwcWallet>()

    @AfterTest
    fun close() = runBlocking<Unit> {
        wallets.forEach { runCatching { it.close() } }
        scope.cancel()
    }

    @Test
    fun paysAnInvoiceAndReportsThePreimage() = runBlocking<Unit> {
        val wallet = connect(balanceSats = 300)
        val invoice = faucetInvoice(sats = 100)

        val paid =
            assertIs<NwcPayOutcome.Settled>(
                wallet.payInvoice(invoice.bolt11, null, invoice.paymentHash)
            )

        assertIs<String>(paid.preimage)
    }

    @Test
    fun aPaidInvoiceIsRefusedDefinitelyNotLeftUncertain() = runBlocking<Unit> {
        // PR-03: a second attempt must be a definite refusal, never "uncertain" (which would
        // leave Rayl waiting) and never paid twice. NWC has no already-paid code (MOB-90, A6).
        val wallet = connect(balanceSats = 300)
        val invoice = faucetInvoice(sats = 100)
        assertIs<NwcPayOutcome.Settled>(
            wallet.payInvoice(invoice.bolt11, null, invoice.paymentHash)
        )

        val second = wallet.payInvoice(invoice.bolt11, null, invoice.paymentHash)

        assertIs<NwcPayOutcome.WalletRejected>(second, "Second attempt was $second")
    }

    @Test
    fun anEmptyWalletIsRefusedDefinitely() = runBlocking<Unit> {
        val wallet = connect(balanceSats = 0)
        val invoice = faucetInvoice(sats = 100)

        val outcome = wallet.payInvoice(invoice.bolt11, null, invoice.paymentHash)

        // MOB-115: Alby Hub's INSUFFICIENT_BALANCE reply over NIP-44 never reaches Rayl, which
        // times out after 15 s and reports the outcome as unknown.
        knownDefect("MOB-115") {
            assertIs<NwcPayOutcome.WalletRejected>(outcome, "Outcome was $outcome")
        }
    }

    /** A new faucet wallet, connected the way the app connects a pasted NWC URI. */
    private suspend fun connect(balanceSats: Long): NwcWallet {
        val uri = post("$FAUCET?balance=$balanceSats").trim()
        val wallet = createNwcWallet(
            secureSettings = TestSecureStringStore(),
            scope = scope,
            isNetworkAvailable = { true },
            startMode = WalletStartMode.Fresh
        )
        wallets += wallet
        wallet.connect(uri, alias = "faucet")
        return wallet
    }

    private class FaucetInvoice(val bolt11: String, val paymentHash: String)

    /** An invoice the faucet itself receives. */
    private fun faucetInvoice(sats: Long): FaucetInvoice {
        val json = Json.parseToJsonElement(post("$FAUCET/make-invoice?amount=$sats")).jsonObject
        return FaucetInvoice(
            bolt11 = json.getValue("invoice").jsonPrimitive.content,
            paymentHash = json.getValue("paymentHash").jsonPrimitive.content
        )
    }

    private fun post(url: String): String {
        val response = http.send(
            HttpRequest.newBuilder(URI(url)).POST(HttpRequest.BodyPublishers.noBody()).build(),
            HttpResponse.BodyHandlers.ofString()
        )
        check(response.statusCode() in 200..299) {
            "Faucet answered ${response.statusCode()}: ${response.body()}"
        }
        return response.body()
    }

    private companion object {
        const val FAUCET = "https://faucet.nwc.dev"
        val http: HttpClient = HttpClient.newHttpClient()
    }
}

/**
 * A check that fails because of an open defect (qa/test-plan.md §10): passes while [block] fails,
 * fails once it passes. Same as the regtest bank's `knownDefect`, which NWC tests don't depend on.
 */
private inline fun knownDefect(issue: String, block: () -> Unit) {
    try {
        block()
    } catch (_: AssertionError) {
        return
    }
    throw AssertionError("$issue seems fixed: remove knownDefect(\"$issue\") and close the issue")
}
